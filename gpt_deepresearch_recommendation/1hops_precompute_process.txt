import os

def _process_spot(center_spot_id, adata, tiles_base_dir, sentences_base_dir):
    """
    Worker function to process a single spot. Encapsulates the logic for one item.
    Returns a sample dictionary on success, None on failure.
    """
    try:
        # a. Get 1-hop neighborhood
        center_idx_global = adata.obs_names.get_loc(center_spot_id)
        neighbor_indices = adata.obsp['spatial_connectivities'][center_idx_global].indices
        neighbor_ids = adata.obs_names[neighbor_indices].tolist()
        all_spot_ids = [center_spot_id] + neighbor_ids
        
        # b. Build local graph structure
        id_to_local_idx = {spot_id: i for i, spot_id in enumerate(all_spot_ids)}
        edge_list = []
        for u_spot_id in all_spot_ids:
            u_global_idx = adata.obs_names.get_loc(u_spot_id)
            u_global_neighbors = adata.obs_names[adata.obsp['spatial_connectivities'][u_global_idx].indices]
            for v_spot_id in u_global_neighbors:
                if v_spot_id in id_to_local_idx:
                    local_idx_u = id_to_local_idx[u_spot_id]
                    local_idx_v = id_to_local_idx[v_spot_id]
                    if local_idx_u < local_idx_v:
                        edge_list.append([local_idx_u, local_idx_v])
        
        # c. Prepare the sample dictionary
        sample = {"__key__": center_spot_id}
        sample["json"] = json.dumps({"num_nodes": len(all_spot_ids), "edge_index": edge_list}).encode('utf-8')

        # d. Dynamically CONSTRUCT paths for all nodes and load data
        for spot_id in all_spot_ids:
            local_idx = id_to_local_idx[spot_id]
            spot_metadata = adata.obs.loc[spot_id]
            
            sample_id = spot_metadata['sample_id']
            row_val = spot_metadata['pxl_row_in_fullres']
            col_val = spot_metadata['pxl_col_in_fullres']
            
            if pd.isna(row_val) or pd.isna(col_val):
                spot_adata_index = adata.obs_names.get_loc(spot_id)
                filename_suffix = f"{spot_adata_index}_0_no_coord"
            else:
                row_coord = int(round(row_val))
                col_coord = int(round(col_val))
                filename_suffix = f"{row_coord}_{col_coord}"
                
            tile_path = os.path.join(tiles_base_dir, f"{sample_id}_tiles", f"{sample_id}_{filename_suffix}.png")
            sentence_path = os.path.join(sentences_base_dir, f"{sample_id}_sentences_hvg", f"{sample_id}_{filename_suffix}.txt")

            with open(tile_path, 'rb') as f_img:
                sample[f"{local_idx}.png"] = f_img.read()
            with open(sentence_path, 'rb') as f_txt:
                sample[f"{local_idx}.txt"] = f_txt.read()

        return sample

    except Exception as e:
        # Log errors at a lower level for parallel execution to avoid clutter
        # logging.debug(f"Skipping {center_spot_id} -> Error: {e}")
        return None


def create_prepackaged_dataset(
    adata_path: str, 
    tiles_base_dir: str,
    sentences_base_dir: str,
    output_dir: str, 
    max_shard_size: int,
    max_workers: int = 16
):
    """
    Main function to perform pre-packaging, now parallelized with ThreadPoolExecutor.

    Args:
        adata_path (str): Path to the master AnnData file.
        tiles_base_dir (str): Base directory containing sample-specific tile folders.
        sentences_base_dir (str): Base directory containing sample-specific sentence folders.
        output_dir (str): Directory to save the output .tar shards.
        max_shard_size (int): Max samples per shard.
        max_workers (int): Number of worker threads for parallel processing.
    """
    logging.info(f"Starting data pre-packaging process with {max_workers} workers.")
    start_time = time.time()
    
    # --- 1. Validation and Setup ---
    os.makedirs(output_dir, exist_ok=True)
    shards_pattern = os.path.join(output_dir, "shard-%06d.tar")
    
    # --- 2. Load AnnData ---
    logging.info(f"Loading AnnData: {adata_path}")
    adata = sc.read_h5ad(adata_path)
    logging.info(f"✅ AnnData loaded. Total spots: {adata.n_obs}")

    required_obs = ['sample_id', 'pxl_col_in_fullres', 'pxl_row_in_fullres']
    if not all(col in adata.obs.columns for col in required_obs):
        raise ValueError(f"AnnData .obs must contain the columns: {required_obs}")
    if 'spatial_connectivities' not in adata.obsp:
        raise ValueError("AnnData must contain 'spatial_connectivities' in .obsp")

    # --- 3. Main Loop (Parallelized) ---
    logging.info(f"Writing shards to: {shards_pattern}")
    error_count = 0
    success_count = 0
    
    with wds.ShardWriter(shards_pattern, maxcount=max_shard_size) as sink:
        with ThreadPoolExecutor(max_workers=max_workers) as executor:
            # Submit all tasks to the executor
            futures = {
                executor.submit(_process_spot, spot_id, adata, tiles_base_dir, sentences_base_dir): spot_id 
                for spot_id in adata.obs.index
            }
            
            # Process results as they complete
            progress_bar = tqdm(as_completed(futures), total=len(futures), desc="Processing Spots", unit="spot")
            for future in progress_bar:
                result = future.result()
                if result is not None:
                    sink.write(result)
                    success_count += 1
                else:
                    spot_id = futures[future]
                    logging.warning(f"Skipping {spot_id} due to an error during processing.")
                    error_count += 1

    # Final summary
    total_time = time.time() - start_time
    logging.info("\n" + "="*50)
    logging.info("✅ Pre-packaging complete!")
    logging.info(f"New dataset saved to: {output_dir}")
    logging.info(f"Total spots processed: {adata.n_obs}")
    logging.info(f"Successfully packaged samples: {success_count}")
    logging.info(f"Samples skipped due to errors: {error_count}")
    logging.info(f"Total time: {total_time / 60:.2f} minutes")
    logging.info("="*50)

DIRECTORIES_TO_SCAN = [
    "/cwStorage/nodecw_group/jijh/hest_output",        # 包含 ..._tiles 文件夹的根目录

def decompress_and_backup_gz_files(root_dirs: list):
    """
    Recursively scans directories, finds .gz files, moves them to a backup
    folder, and leaves the decompressed version in the original location.
    """
    print(f"🔍 开始扫描目录: {root_dirs}")
    total_found = 0
    total_processed = 0
    
    for root_dir in root_dirs:
        if not os.path.isdir(root_dir):
            print(f"⚠️ 警告: 目录不存在，跳过: {root_dir}")
            continue
            
        # 使用 os.walk 遍历所有子目录和文件
        file_list = []
        for dirpath, _, filenames in os.walk(root_dir):
            for filename in filenames:
                if filename.endswith(".gz"):
                    file_list.append(os.path.join(dirpath, filename))
        
        if not file_list:
            print(f"✅ 在 '{root_dir}' 中没有找到 .gz 文件。")
            continue

        print(f"在 '{root_dir}' 中找到 {len(file_list)} 个 .gz 文件，开始处理...")
        total_found += len(file_list)
        
        with tqdm(total=len(file_list), desc=f"处理 {os.path.basename(root_dir)}") as pbar:
            for gz_path in file_list:
                try:
                    # 1. 确定原始文件名和备份路径
                    original_path = gz_path[:-3]  # 去掉 ".gz" 后缀
                    backup_dir = os.path.dirname(gz_path) + "_gz_backup"
                    os.makedirs(backup_dir, exist_ok=True)
                    backup_path = os.path.join(backup_dir, os.path.basename(gz_path))
                    
                    # 2. 移动 .gz 文件到备份目录
                    shutil.move(gz_path, backup_path)
                    
                    # 3. 从备份位置解压到原始位置
                    with gzip.open(backup_path, 'rb') as f_in:
                        with open(original_path, 'wb') as f_out:
                            shutil.copyfileobj(f_in, f_out)
                    
                    total_processed += 1
                except Exception as e:
                    print(f"\n❌ 处理文件 '{gz_path}' 时出错: {e}")
                finally:
                    pbar.update(1)

    print("\n" + "="*50)
    print("✅ 全部处理完成！")
    print(f"总共找到 .gz 文件: {total_found}")
    print(f"成功处理并恢复的文件: {total_processed}")
    print("="*50)

# --- 运行修复工具 ---
decompress_and_backup_gz_files(DIRECTORIES_TO_SCAN)

# --- INPUTS ---

def setup_logging():
    """Configures logging for the script."""
    logging.basicConfig(
        level=logging.INFO,
        format='%(asctime)s [%(levelname)s] - %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

def process_and_encode_sample(sample, model, image_preprocessor, tokenizer, device):
    """
    Worker function: Takes one raw data sample, encodes its contents, 
    and returns a new sample dictionary with embeddings.
    Now returns an error message on failure for detailed reporting.
    """
    try:
        key = sample['__key__']
        
        # 1. Decode metadata and raw data
        metadata = json.loads(sample['json'])
        num_nodes = metadata['num_nodes']
        
        images_to_process = []
        texts_to_tokenize = []
        for i in range(num_nodes):
            images_to_process.append(Image.open(io.BytesIO(sample[f'{i}.png'])).convert("RGB"))
            texts_to_tokenize.append(sample[f'{i}.txt'].decode('utf-8'))

        # 2. Preprocess data into tensors
        image_input = torch.stack([image_preprocessor(img) for img in images_to_process])
        text_input = tokenizer(texts_to_tokenize)

        # 3. Encode on GPU
        with torch.no_grad():
            image_input = image_input.to(device)
            text_input = text_input.to(device)
            
            image_embeddings = model.encode_image(image_input).cpu()
            text_embeddings = model.encode_text(text_input).cpu()

        # 4. Construct the new sample with serialized tensor embeddings
        output_sample = {"__key__": key, "json": sample['json']}
        for i in range(num_nodes):
            img_buffer = io.BytesIO()
            torch.save(image_embeddings[i], img_buffer)
            output_sample[f"{i}.image.pth"] = img_buffer.getvalue()

            txt_buffer = io.BytesIO()
            torch.save(text_embeddings[i], txt_buffer)
            output_sample[f"{i}.text.pth"] = txt_buffer.getvalue()
        
        return output_sample, None
    
    except Exception as e:
        error_type = type(e).__name__
        return None, f"Sample {sample.get('__key__', 'UNKNOWN_KEY')} failed with {error_type}: {e}"


def create_embedding_dataset():
    """Main function to drive the pre-computation and sharding process."""
    setup_logging()
    
    logging.info("=" * 80)
    logging.info("🚀 Starting Embedding Pre-computation Script")
    logging.info("=" * 80)
    logging.info("CONFIGURATION:")
    logging.info(f"  - Input Raw Shards:     {INPUT_SHARDS_DIR}")
    logging.info(f"  - Output Embedding Shards: {OUTPUT_EMBEDDING_DIR}")
    logging.info(f"  - Base Model Path:        {BASE_MODEL_PATH}")
    logging.info(f"  - Model Architecture:     {MODEL_NAME}")
    logging.info(f"  - Max Samples per Shard:  {MAX_SAMPLES_PER_SHARD}")
    logging.info(f"  - Max CPU Workers:        {MAX_WORKERS}")
    logging.info("-" * 80)
    
    start_time = time.time()
    
    # --- 1. Setup paths and model ---
    os.makedirs(OUTPUT_EMBEDDING_DIR, exist_ok=True)
    if any(os.scandir(OUTPUT_EMBEDDING_DIR)):
        logging.warning(f"⚠️ Output directory '{OUTPUT_EMBEDDING_DIR}' is not empty. Files may be overwritten.")
        
    shards_pattern = os.path.join(OUTPUT_EMBEDDING_DIR, "shard-%06d.tar")
    
    device = "cuda" if torch.cuda.is_available() else "cpu"
    logging.info(f"🔌 Using device: {device}")

    logging.info(f"🔧 Loading OmiCLIP model '{MODEL_NAME}'...")
    model, _, image_preprocessor = open_clip.create_model_and_transforms(
        MODEL_NAME,
        pretrained=BASE_MODEL_PATH,
        device=device,
        output_dict=True,
    )
    model.eval()
    tokenizer = open_clip.get_tokenizer(MODEL_NAME)
    logging.info("✅ Model and preprocessors loaded successfully.")

    input_shards = sorted(glob.glob(os.path.join(INPUT_SHARDS_DIR, "*.tar")))
    if not input_shards:
        logging.error(f"❌ No input shards found at: {os.path.join(INPUT_SHARDS_DIR, '*.tar')}")
        return
        
    logging.info(f"📚 Found {len(input_shards)} input shards to process.")

    # --- 2. Run the processing and writing loop ---
    logging.info(f"✍️  Writing new embedding shards to: {shards_pattern}")
    error_count = 0
    success_count = 0
    error_types = collections.Counter()
    last_log_time = time.time()
    
    with wds.ShardWriter(shards_pattern, maxcount=MAX_SAMPLES_PER_SHARD) as sink:
        with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
            # Create a wds pipeline for each shard to better track progress
            shard_pbar = tqdm(input_shards, desc="Overall Shard Progress", unit="shard")
            for shard_path in shard_pbar:
                shard_name = os.path.basename(shard_path)
                shard_pbar.set_postfix_str(f"Processing {shard_name}")

                # Create a dataset for the current shard
                shard_dataset = wds.WebDataset(shard_path)
                
                # Submit jobs for the current shard
                future_to_key = {
                    executor.submit(process_and_encode_sample, sample, model, image_preprocessor, tokenizer, device): sample['__key__']
                    for sample in shard_dataset
                }

                # Process results as they complete
                for future in as_completed(future_to_key):
                    result, error_msg = future.result()
                    if result is not None:
                        sink.write(result)
                        success_count += 1
                    else:
                        error_count += 1
                        # Log detailed error and count its type
                        if error_msg:
                            error_type = error_msg.split(':')[1].split(' ')[1] if ':' in error_msg else "UnknownError"
                            error_types[error_type] += 1
                            logging.debug(error_msg) # Log full error message in debug mode
                
                # Performance logging
                current_time = time.time()
                if current_time - last_log_time > LOG_INTERVAL_SECONDS:
                    elapsed = current_time - start_time
                    graphs_per_sec = success_count / elapsed if elapsed > 0 else 0
                    logging.info(
                        f"  -> Progress: {success_count} graphs processed. "
                        f"Speed: {graphs_per_sec:.2f} graphs/sec. "
                        f"Errors: {error_count}."
                    )
                    last_log_time = current_time

    # --- 3. Final summary ---
    total_time = time.time() - start_time
    final_rate = success_count / total_time if total_time > 0 else 0
    logging.info("\n" + "="*80)
    logging.info("🏁 Pre-computation Complete!")
    logging.info(f"  - Output Directory: {OUTPUT_EMBEDDING_DIR}")
    logging.info(f"  - Successfully Packaged Subgraphs: {success_count}")
    logging.info(f"  - Subgraphs Skipped due to Errors: {error_count}")
    if error_count > 0:
        logging.info("  - Error Type Breakdown:")
        for err_type, count in error_types.items():
            logging.info(f"    - {err_type}: {count}")
    logging.info(f"  - Total Time: {total_time:.2f} seconds ({total_time / 60:.2f} minutes)")
    logging.info(f"  - Average Throughput: {final_rate:.2f} graphs/sec")
    logging.info("="*80)
    logging.info("Next Step: Update your training script's --train-data path to point to the new embedding shards.")


if __name__ == '__main__':
    create_embedding_dataset()
