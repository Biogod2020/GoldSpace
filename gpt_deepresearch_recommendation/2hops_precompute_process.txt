def get_k_hop_neighborhood(adjacency_matrix: csr_matrix, start_node_idx: int, k: int) -> list[int]:
    """高效的BFS函数，用于查找k-hop邻域"""
    if k == 0: return [start_node_idx]
    visited = {start_node_idx}
    queue = deque([(start_node_idx, 0)])
    neighborhood = [start_node_idx]
    while queue:
        current_node, level = queue.popleft()
        if level >= k: continue
        start_ptr, end_ptr = adjacency_matrix.indptr[current_node], adjacency_matrix.indptr[current_node + 1]
        neighbors = adjacency_matrix.indices[start_ptr:end_ptr]
        for neighbor in neighbors:
            if neighbor not in visited:
                visited.add(neighbor)
                neighborhood.append(neighbor)
                queue.append((neighbor, level + 1))
    return neighborhood

        spot_metadata = adata.obs.loc[spot_id]

            col_coord = int(round(spot_metadata['pxl_col_in_fullres']))
            filename_suffix = f"{row_coord}_{col_coord}"

            sentence_text = f_txt.read()
            
        return spot_id, {"tile": tile_image, "sentence": sentence_text}
    except Exception as e:
        return spot_id, e

# --- 工具函数 (如果需要，可以从前面的单元格复制过来) ---
def get_k_hop_neighborhood(adjacency_matrix: csr_matrix, start_node_idx: int, k: int) -> list[int]:
    """高效的BFS函数，用于查找k-hop邻域"""
    if k == 0: return [start_node_idx]
    visited = {start_node_idx}
    queue = deque([(start_node_idx, 0)])
    neighborhood = [start_node_idx]
    while queue:
        current_node, level = queue.popleft()
        if level >= k: continue
        start_ptr, end_ptr = adjacency_matrix.indptr[current_node], adjacency_matrix.indptr[current_node + 1]
        neighbors = adjacency_matrix.indices[start_ptr:end_ptr]
        for neighbor in neighbors:
            if neighbor not in visited:
                visited.add(neighbor)
                neighborhood.append(neighbor)
                queue.append((neighbor, level + 1))
    return neighborhood

def assemble_and_write_shards(adata, adj_matrix, embedding_cache, output_dir, k_hops, max_samples_per_shard, max_workers):
    """
    Assembles subgraphs using pre-computed embeddings and writes them to WebDataset shards.
    This is the final step of the preprocessing pipeline.
    """
    logging.info(f"🔥 Phase 3: Assembling subgraphs and writing to WebDataset shards...")
    start_time_phase3 = time.time()
    
    # 1. Setup output directory and shard writer
    os.makedirs(output_dir, exist_ok=True)
    shards_pattern = os.path.join(output_dir, "shard-%06d.tar")
    
    # 2. Define the worker function for parallel processing
    def _assemble_subgraph_worker(center_spot_id: str):
        """
        Worker function for a single center spot. It performs graph building and
        embedding assembly in memory and returns a sample dict for the shard writer.
        """
        try:
            # a. Get k-hop neighborhood using the pre-loaded adjacency matrix
            center_node_idx = adata.obs_names.get_loc(center_spot_id)
            global_indices = get_k_hop_neighborhood(adj_matrix, center_node_idx, k_hops)
            all_spot_ids = adata.obs_names[global_indices].tolist()
            
            # b. Build local graph structure
            global_id_to_local_idx = {sid: i for i, sid in enumerate(all_spot_ids)}
            local_edge_index = []
            for local_u_idx, u_id in enumerate(all_spot_ids):
                u_global_idx = adata.obs_names.get_loc(u_id)
                start, end = adj_matrix.indptr[u_global_idx], adj_matrix.indptr[u_global_idx + 1]
                for v_global_idx in adj_matrix.indices[start:end]:
                    v_id = adata.obs_names[v_global_idx]
                    if v_id in global_id_to_local_idx:
                        local_v_idx = global_id_to_local_idx[v_id]
                        if local_u_idx < local_v_idx:
                            local_edge_index.append([local_u_idx, local_v_idx])
            
            # c. Collect pre-computed embeddings from the cache
            # This is extremely fast as it's just a dictionary lookup
            img_embeds = [embedding_cache[sid]['image_embed'] for sid in all_spot_ids]
            txt_embeds = [embedding_cache[sid]['text_embed'] for sid in all_spot_ids]
            
            # d. Package into the SOTA format: one .pth file per sample
            embeddings_dict = {
                'image': torch.stack(img_embeds),  # Shape: [num_nodes, embed_dim]
                'text': torch.stack(txt_embeds)    # Shape: [num_nodes, embed_dim]
            }
            emb_buf = io.BytesIO()
            torch.save(embeddings_dict, emb_buf)
            
            # e. Assemble the final sample for webdataset
            sample = {
                "__key__": center_spot_id,
                "json": json.dumps({"num_nodes": len(all_spot_ids), "edge_index": local_edge_index}).encode('utf-8'),
                "embeddings.pth": emb_buf.getvalue()
            }
            return sample, None
        
        except Exception as e:
            return None, f"Spot '{center_spot_id}' failed with {type(e).__name__}: {e}"

    # 3. Main parallel processing loop
    success_count = 0
    error_count = 0
    with wds.ShardWriter(shards_pattern, maxcount=max_samples_per_shard) as sink:
        with ThreadPoolExecutor(max_workers=max_workers) as executor:
            # Submit all spots for processing
            future_to_spot = {
                executor.submit(_assemble_subgraph_worker, spot_id): spot_id 
                for spot_id in adata.obs.index
            }
            
            pbar = tqdm(as_completed(future_to_spot), total=len(future_to_spot), desc="Assembling & Writing Shards")
            for future in pbar:
                sample, error_msg = future.result()
                if sample:
                    sink.write(sample)
                    success_count += 1
                else:
                    error_count += 1
                    logging.debug(error_msg) # Log errors for failed spots

    # 4. Final summary
    logging.info(f"✅ Phase 3 finished in {(time.time() - start_time_phase3) / 60:.2f} minutes.")
    logging.info(f"Successfully created {success_count} samples. Failed: {error_count}.")
    logging.info(f"Find your final embedding shards at: {output_dir}")

def assemble_and_write_shards_DEBUG(adata, adj_matrix, embedding_cache, output_dir, k_hops, max_samples_per_shard):

    shards_pattern = os.path.join(output_dir, "shard-%06d.tar")
    
    def _assemble_subgraph_worker(center_spot_id: str):
        """
        Worker function for a single center spot.
        """
        # --- MODIFICATION START ---
        # Catch specific errors to provide better debug information.
        try:
            # a. Get k-hop neighborhood
            center_node_idx = adata.obs_names.get_loc(center_spot_id)
            global_indices = get_k_hop_neighborhood(adj_matrix, center_node_idx, k_hops)
            all_spot_ids = adata.obs_names[global_indices].tolist()
            
            # b. Build local graph structure
            global_id_to_local_idx = {sid: i for i, sid in enumerate(all_spot_ids)}
            local_edge_index = []
            for local_u_idx, u_id in enumerate(all_spot_ids):
                u_global_idx = adata.obs_names.get_loc(u_id)
                start, end = adj_matrix.indptr[u_global_idx], adj_matrix.indptr[u_global_idx + 1]
                for v_global_idx in adj_matrix.indices[start:end]:
                    v_id = adata.obs_names[v_global_idx]
                    if v_id in global_id_to_local_idx:
                        local_v_idx = global_id_to_local_idx[v_id]
                        if local_u_idx < local_v_idx:
                            local_edge_index.append([local_u_idx, local_v_idx])
            
            # c. CRITICAL STEP: Collect pre-computed embeddings from the cache
            # This is where a KeyError is most likely to occur.
            img_embeds = [embedding_cache[sid]['image_embed'] for sid in all_spot_ids]
            txt_embeds = [embedding_cache[sid]['text_embed'] for sid in all_spot_ids]
            
            # d. Package into SOTA format
            embeddings_dict = {
                'image': torch.stack(img_embeds),
                'text': torch.stack(txt_embeds)
            }
            emb_buf = io.BytesIO()
            torch.save(embeddings_dict, emb_buf)
            
            # e. Assemble the final sample
            sample = {
                "__key__": center_spot_id,
                "json": json.dumps({"num_nodes": len(all_spot_ids), "edge_index": local_edge_index}).encode('utf-8'),
                "embeddings.pth": emb_buf.getvalue()
            }
            return sample, None
        
        except KeyError as e:
            # This is the most likely error. It means a spot in the subgraph was not found in the embedding cache.
            return None, f"FATAL KeyError: Spot ID '{e.args[0]}' is required for the subgraph of '{center_spot_id}', but it was not found in the embedding_cache. This indicates a failure in the embedding generation phase."
        except Exception as e:
            return None, f"FATAL Unexpected Error for spot '{center_spot_id}': {type(e).__name__} - {e}"
        # --- MODIFICATION END ---

def assemble_and_write_shards_ROBUST(adata, adj_matrix, embedding_cache, output_dir, k_hops, max_samples_per_shard, max_workers):
    """
    ROBUST PRODUCTION VERSION: Assembles subgraphs in parallel and logs all failures
    without stopping the entire process.
    """
    logging.info("🔥 Phase 3 (ROBUST MODE): Starting parallel subgraph assembly with CLEANED data...")
    start_time_phase3 = time.time()
    
    # 清理输出目录，确保从头开始生成
    if os.path.exists(output_dir):
        import shutil
        shutil.rmtree(output_dir)
        logging.info(f"Cleaned up previous output directory: {output_dir}")
    os.makedirs(output_dir, exist_ok=True)
    
    shards_pattern = os.path.join(output_dir, "shard-%06d.tar")
    
    # Worker function to assemble one sample
    def _assemble_subgraph_worker(center_spot_id: str):
        try:
            center_node_idx = adata.obs_names.get_loc(center_spot_id)
            global_indices = get_k_hop_neighborhood(adj_matrix, center_node_idx, k_hops)
            all_spot_ids = adata.obs_names[global_indices].tolist()
            
            global_id_to_local_idx = {sid: i for i, sid in enumerate(all_spot_ids)}
            local_edge_index = []
            for local_u_idx, u_id in enumerate(all_spot_ids):
                u_global_idx = adata.obs_names.get_loc(u_id)
                start, end = adj_matrix.indptr[u_global_idx], adj_matrix.indptr[u_global_idx + 1]
                for v_global_idx in adj_matrix.indices[start:end]:
                    v_id = adata.obs_names[v_global_idx]
                    if v_id in global_id_to_local_idx:
                        local_v_idx = global_id_to_local_idx[v_id]
                        if local_u_idx < local_v_idx:
                            local_edge_index.append([local_u_idx, local_v_idx])
            
            # This step will no longer fail
            img_embeds = [embedding_cache[sid]['image_embed'] for sid in all_spot_ids]
            txt_embeds = [embedding_cache[sid]['text_embed'] for sid in all_spot_ids]
            
            embeddings_dict = {'image': torch.stack(img_embeds), 'text': torch.stack(txt_embeds)}
            emb_buf = io.BytesIO()
            torch.save(embeddings_dict, emb_buf)
            
            sample = {
                "__key__": center_spot_id,
                "json": json.dumps({"num_nodes": len(all_spot_ids), "edge_index": local_edge_index}).encode('utf-8'),
                "embeddings.pth": emb_buf.getvalue()
            }
            return sample, None
        except Exception as e:
            # This is now a fallback, we don't expect it to be triggered
            return None, f"Unexpected error for spot '{center_spot_id}': {type(e).__name__} - {e}"

    # Main parallel processing loop
    failed_spots = {}
